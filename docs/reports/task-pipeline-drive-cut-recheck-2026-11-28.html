Appendix: the two-stage chain script

// One two-stage chain, end to end, through the driver's only public entry.
import assert from "node:assert/strict";
import { mkdirSync, mkdtempSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import path from "node:path";

import { parseBundle } from "./lib/pipeline-bundle.js";
import { settleStageRun } from "./lib/task-pipeline.js";
import { driveTaskPipelines, STARTING_AT } from "./lib/task-pipeline-drive.js";

const CLOCK = "2026-11-03T16:00:00.000Z";
const LATER = "2026-11-03T16:10:00.000Z";

const stage = (n) => ({
  id: `S${n}`, title: `Stage ${n}`, prompt: `Work stage ${n}.`,
  artifact: { id: `artifact-${n}`, path: `out/${n}.json`, schema: "out/1", writer: `worker-${n}` },
});
const doc = {
  id: "two", title: "Two", stages: [stage(1), stage(2)], gates: [],
  handoffs: [{ id: "h1", from: "S1", to: "S2", artifact: "artifact-1", kind: "sequence" }],
};
const parsed = parseBundle({ bundleVersion: 1, name: "chain", pipelines: [doc] }, { file: "pipeline-bundle.json" });
assert.equal(parsed.ok, true, parsed.reason);

const dir = mkdtempSync(path.join(tmpdir(), "wag-chain-"));
mkdirSync(path.join(dir, "out"), { recursive: true });
writeFileSync(path.join(dir, "out", "1.json"), JSON.stringify({ stage: 1 }));

const row = {
  id: 1, title: "carry the contract", detail: "", state: "running", thread: "T1", threads: ["T1"],
  worktree: dir, project: "agent-worktree-gui", files: [], attempts: 0, reason: "", hold: null,
  pipelineRun: { pipeline: "two", stage: "S1", enteredAt: "2026-11-03T15:00:00.000Z" },
};
const jobs = [{ id: "j1", thread: "T1", status: "done", startedAt: "2026-11-03T15:00:00.000Z", endedAt: "2026-11-03T15:30:00.000Z" }];

// pass 1: S1 settled, S2 asked for
const one = driveTaskPipelines([row], { bundle: parsed.bundle, jobs, isLive: () => false, at: CLOCK, unattended: true });
const afterOne = one.tasks[0];
assert.equal(afterOne.state, "queued");
assert.equal(afterOne.pipelineRun.stage, "S2");
assert.equal(afterOne.pipelineRun.crossings.h1, 1);
assert.deepEqual(afterOne.pipelineRun.visits, { S1: 1, S2: 1 });
assert.equal(one.starts.length, 1);
assert.equal(one.starts[0].stage, "S2");
assert.equal(one.starts[0].position, 2);
assert.equal(one.starts[0].count, 2);
assert.equal(one.starts[0].target, dir);
assert.match(one.starts[0].prompt, /Work stage 2\./);
assert.equal(afterOne.pipelineRun[STARTING_AT], CLOCK);

// the link: the record fills in the thread and job and leaves the entry open
writeFileSync(path.join(dir, "out", "2.json"), JSON.stringify({ stage: 2 }));
const linked = settleStageRun(
  { ...afterOne, state: "running", thread: "T2", threads: ["T1", "T2"] },
  { stage: "S2", visit: 1, thread: "T2", job: "j2", tree: dir },
);

// pass 2: S2 settled, the pipeline is over
const two = driveTaskPipelines([linked], {
  bundle: parsed.bundle,
  jobs: [...jobs, { id: "j2", thread: "T2", status: "done", startedAt: LATER, endedAt: LATER }],
  isLive: () => false, at: LATER, unattended: true,
});
const afterTwo = two.tasks[0];
assert.equal(two.starts.length, 0);
assert.equal(afterTwo.state, "done");
assert.equal(afterTwo.doneBy, "panel");
assert.equal(afterTwo.pipelineRun.waitingFor, null);
assert.match(afterTwo.reason, /pipeline two is over at S2/);
assert.match(afterTwo.reason, /artifact "artifact-2"/);
assert.match(afterTwo.reason, /not a PR to land/);
assert.deepEqual(afterTwo.pipelineRun.stageRuns.map((r) => `${r.id}:${r.status}`), ["two/S1#1:done", "two/S2#1:done"]);

rmSync(dir, { recursive: true, force: true });
console.log("CHAIN PASSED");
